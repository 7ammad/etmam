+/**
+ * Generate a human-readable summary explaining WHY the score is what it is.
+ * Based on the breakdown dimensions - highlights strongest and weakest areas.
+ */
+function generateScoreSummary(
+  score: number,
+  breakdown: { budget_fit: number; technical_fit: number; timeline_fit: number; strategic_fit: number; risk_score: number },
+  recommendation: Recommendation
+): string {
+  const dims = [
+    { key: 'budget_fit', label: 'Budget Fit', value: breakdown.budget_fit },
+    { key: 'technical_fit', label: 'Technical Fit', value: breakdown.technical_fit },
+    { key: 'timeline_fit', label: 'Timeline', value: breakdown.timeline_fit },
+    { key: 'strategic_fit', label: 'Strategic Fit', value: breakdown.strategic_fit },
+    { key: 'risk_score', label: 'Risk', value: breakdown.risk_score },
+  ].sort((a, b) => b.value - a.value)
+
+  const strongest = dims[0]
+  const weakest = dims[dims.length - 1]
+
+  // Build summary explaining the score
+  const parts: string[] = []
+
+  if (recommendation === 'qualified') {
+    parts.push(`Strong overall score of ${score}.`)
+  } else if (recommendation === 'conditional') {
+    parts.push(`Moderate score of ${score}.`)
+  } else {
+    parts.push(`Low score of ${score}.`)
+  }
+
+  // Explain strongest dimension
+  if (strongest.value >= 70) {
+    parts.push(`${strongest.label} is strong (${strongest.value}).`)
+  }
+
+  // Explain weakest dimension if it's dragging the score down
+  if (weakest.value < 50) {
+    parts.push(`${weakest.label} is a concern (${weakest.value}).`)
+  }
+
+  return parts.join(' ')
